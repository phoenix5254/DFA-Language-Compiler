
<program> := <dfa_declaration>

<dfa_declaration>:= <identifier> '=' (<formate_a> | <format_b>)


#Tuple Multi-Line Definition
formate_a:= <LETTERS> '=' '(' <all_states_symbols> ',' <input_symbols> ',' <transition_symbol> ',' <start_state_symbol> ',' <final_states_symbols> ) '\n'
                            <transition_function> '\n'            

#Tuple in-line definition
formate_b:= <LETTERS> '=''(' <set_definition> ',' <set_definition> ',' <transition_symbol> ',' <start_s> ',' <final_states> ')' 
                            '\n' <transition_function> '\n'



<all_states>:= <LETTERS> '=' set_definition 
<transition_function>:= <transition_symbol> ( '('<identifier> ',' <identifier> ')' ('='|'->') <identifier> (',')* )+
                    | '('<transition_symbol>  '*' <identifier> ('='|'->') <identifier> (',')* )+

<start_s>:= <identifier> 
<final_states>:= <set_definition>
<all_states_symbols>:= <LETTERS> '=' <set_definition> '\n'
<input_symbols>:= <LETTERS> '=' <set_definition> '\n'
<start_state_symbol>:= <LETTERS> '=' <identifier> '\n'
<final_states_symbols>:= <LETTERS> '=' <set_definition> '\n'




#Core Elements
<set_definition>:= '{' <identifier>* '}'
<identifier>:= <ALPHANUMERIC>
<transition_symbol>:= 'delta' | 'DELTA' | 'δ'
<ALPHANUMERIC> := [a-zA-Z][a-zA-Z0-9_]
<LETTERS> := [a-zA-Z]
<NUMBERS> := [0-9]
